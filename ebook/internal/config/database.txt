package config

import (
	"database/sql"

	_ "github.com/go-sql-driver/mysql"
)


// Gestion de Base de Datos

func NewMySQL(
	dsn string,
) (*sql.DB, error) {

	db, err := sql.Open(
		"mysql",
		dsn,
	)

	if err != nil {
		return nil, err
	}

	if err := db.Ping(); err != nil {
		return nil, err
	}

	return db, nil
}
